# Ridge archers who tip their arrows with volcanic glass, after Drogg the Stone Fletcher.
# Patterned on vanilla a_early_longbowmen / a_longbowmen: a culture levy that hits harder.

	is_special = yes
	upgrades_to = a_ga_blackglass_crossbows

	strength_damage_done = 0.10
	# Scrap armour and brittle points.
	strength_damage_taken = 0.10

	construction_demand = archer_infantry_construction

	combat = {
		hills = 0.15
		mountains = 0.10

		multiply = 0.01
	}

	location_potential = {
		ga_obsidian_fletchers_location_trigger = yes
	}

	gfx_tags = { archer_tag light_tag }
}
