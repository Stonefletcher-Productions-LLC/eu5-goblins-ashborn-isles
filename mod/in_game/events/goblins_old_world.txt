namespace = ga_old_world
# The Old World Reacts: how Iberian and Catholic courts meet the goblins, and how
# every Ashborn crown answers them. Paced by the goblins_old_world on_actions.

# Foreign Catholic courts that received goblin first contact debate the goblins.
ga_old_world.1 = {
    type = country_event outcome = neutral
    title = ga_old_world.1.title desc = ga_old_world.1.desc
    image = "gfx/interface/illustrations/event/ashborn/old_world_demons.dds"
    trigger = { ga_is_goblin = no has_variable = ga_received_goblin_first_contact religion = religion:catholic }
    option = { name = ga_old_world.1.a ai_chance = { factor = 2 }
        set_variable = { name = ga_owr_souls value = yes }
        custom_tooltip = ga_old_world.1.a_tt
        hidden_effect = { every_country = { limit = { ga_is_goblin = yes } save_scope_as = ga_owr_goblin root = { add_opinion = { target = scope:ga_owr_goblin modifier = ga_owr_new_souls } } } }
        add_country_modifier = { modifier = ga_owr_mission_to_the_isles years = 10 mode = replace }
    }
    option = { name = ga_old_world.1.b ai_chance = { factor = 1 }
        set_variable = { name = ga_owr_demons value = yes }
        custom_tooltip = ga_old_world.1.b_tt
        hidden_effect = { every_country = { limit = { ga_is_goblin = yes } save_scope_as = ga_owr_goblin root = { add_opinion = { target = scope:ga_owr_goblin modifier = ga_owr_spawn_of_the_pit } } } }
        add_country_modifier = { modifier = ga_owr_watch_on_the_coast years = 10 mode = replace }
    }
}
