ga_mainland.1 = {
    type = country_event
    outcome = neutral
    title = ga_mainland.1.title
    desc = ga_mainland.1.desc
    trigger = { ga_is_goblin = yes ga_holds_mainland = yes }
    image = "gfx/interface/illustrations/event/ashborn/eastern_harbor.dds"
    immediate = { set_variable = { name = ga_ms_landed value = yes } }
    option = { name = ga_mainland.1.a ai_chance = { factor = 3 }
        add_country_modifier = { modifier = ga_ms_first_landing years = 10 mode = replace }
        random_owned_location = { limit = { ga_mainland_location = yes } add_pop = { culture = root.culture religion = root.religion type = pop_type:peasants size = 3 } }
        root.capital ?= {
            ordered_pop = {
                limit = { culture = { has_culture_group = culture_group:cm_goblin_group } pop_size > 9 }
                order_by = pop_size
                max = 1
                add_pop_size = -3
            }
        }
    }
    option = { name = ga_mainland.1.b ai_chance = { factor = 2 }
        add_prestige = 10
        random_owned_location = { limit = { ga_mainland_location = yes } add_location_modifier = { modifier = ga_ms_stockade years = 20 mode = replace } }
    }
}
